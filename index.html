Custom Python Plugins

Preview code snippets for custom widgets included in the plugin package.

plugins/crosshair.py 
import tkinter as tk

class HUDCrosshairPlugin:
    def __init__(self, root):
        self.root = root
        self.root.title("HUDForge - Custom Crosshair")
        self.root.attributes("-topmost", True)
        self.root.overrideredirect(True)
        self.root.config(bg="#000000")
        self.root.wm_attributes("-transparentcolor", "#000000")

        self.canvas = tk.Canvas(root, width=100, height=100, bg="#000000", highlightthickness=0)
        self.canvas.pack()

        # Draw neon cyan crosshair overlay
        self.canvas.create_line(50, 35, 50, 65, fill="#00f0ff", width=2)
        self.canvas.create_line(35, 50, 65, 50, fill="#00f0ff", width=2)

if __name__ == "__main__":
    app = tk.Tk()
    plugin = HUDCrosshairPlugin(app)
    app.mainloop()
